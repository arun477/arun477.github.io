{{- /* A day's items, checked and in reading order: spikes first, then the agent's own order. Each item gains "day" (its page), "kind" (defaults to news) and "source" (falls back to the link's host). A malformed item or day fails the build, so a bad push never reaches the live site. */ -}} {{- $kinds := partial "news/kinds.html" . -}} {{- $spikes := slice }}{{ $rest := slice -}} {{- range $i, $it := .Params.items -}} {{- range slice "title" "url" "summary" }}{{ if not (index $it .) }}{{ errorf "feed: %s, item %d has no %q" $.File.Path (add $i 1) . }}{{ end }}{{ end -}} {{- if not (findRE `^https?://` $it.url) }}{{ errorf "feed: %s, item %d: url must start with http(s)://" $.File.Path (add $i 1) }}{{ end -}} {{- $kind := $it.kind | default "news" -}} {{- if not (index $kinds $kind) }}{{ errorf "feed: %s, item %d: kind %q is not one of %s" $.File.Path (add $i 1) $kind "news, blog, paper, release" }}{{ end -}} {{- $src := $it.source | default (replaceRE `^www\.` "" (urls.Parse $it.url).Host) -}} {{- $it = merge $it (dict "day" $ "kind" $kind "source" $src) -}} {{- if $it.spike }}{{ $spikes = $spikes | append $it }}{{ else }}{{ $rest = $rest | append $it }}{{ end -}} {{- end -}} {{- if not .Params.items }}{{ errorf "feed: %s has no items" .File.Path }}{{ end -}} {{- /* Limits that keep the feed signal, and no free-form body: agent text only ever reaches the page as escaped fields. */ -}} {{- if gt (len .Params.items) 20 }}{{ errorf "feed: %s has %d items; the limit is 20" .File.Path (len .Params.items) }}{{ end -}} {{- if trim .RawContent " \n\r\t" }}{{ errorf "feed: %s has body text; put everything in front matter fields" .File.Path }}{{ end -}} {{- $all := $spikes | append $rest }}{{ return $all -}}
