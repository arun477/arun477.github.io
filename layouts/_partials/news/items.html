{{- /* A day's items, checked and in reading order: spikes first, then the agent's own order. Each item gains "day" (its page) and "source" (falls back to the link's host). A malformed item or day fails the build, so a bad push never reaches the live site. */ -}} {{- $spikes := slice }}{{ $rest := slice -}} {{- range $i, $it := .Params.items -}} {{- range slice "title" "url" "summary" }}{{ if not (index $it .) }}{{ errorf "news: %s, item %d has no %q" $.File.Path (add $i 1) . }}{{ end }}{{ end -}} {{- if not (findRE `^https?://` $it.url) }}{{ errorf "news: %s, item %d: url must start with http(s)://" $.File.Path (add $i 1) }}{{ end -}} {{- $src := $it.source | default (replaceRE `^www\.` "" (urls.Parse $it.url).Host) -}} {{- $it = merge $it (dict "day" $ "source" $src) -}} {{- if $it.spike }}{{ $spikes = $spikes | append $it }}{{ else }}{{ $rest = $rest | append $it }}{{ end -}} {{- end -}} {{- if not .Params.items }}{{ errorf "news: %s has no items" .File.Path }}{{ end -}} {{- /* Limits that keep the feed a brief, and no free-form body: agent text only ever reaches the page as escaped fields. */ -}} {{- if gt (len .Params.items) 12 }}{{ errorf "news: %s has %d items; the limit is 12" .File.Path (len .Params.items) }}{{ end -}} {{- if gt (len $spikes) 2 }}{{ errorf "news: %s has %d spikes; the limit is 2" .File.Path (len $spikes) }}{{ end -}} {{- if trim .RawContent " \n\r\t" }}{{ errorf "news: %s has body text; put everything in front matter fields" .File.Path }}{{ end -}} {{- $all := $spikes | append $rest }}{{ return $all -}}
